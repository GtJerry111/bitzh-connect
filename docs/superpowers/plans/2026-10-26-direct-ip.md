# 直连 IP（不走校园 VPN）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** VPN 保持连接时，让用户指定的公司 IPv4 走物理网关，而学校流量继续走校园 VPN；提供可配置、可校验、可观测、可清理的 UI。

**Architecture:** 不改 zju-connect 内核。App 把用户配置的 IPv4 列表传给特权 helper；helper 在启动内核前给每个 IP 添加 `/32` 路由到当前默认物理网关，并在停止/内核退出时删除。UI 分为高级设置卡片、菜单栏状态行、主窗口失败警告条；配置状态与当前生效状态分离，改配置通过“稍后 / 现在重连”应用。

**Tech Stack:** Python 3.11, PySide6, pytest/pytest-qt, macOS `route`，现有特权 helper（Unix socket + JSON）。

# app/utils/direct_ip.py
"""直连 IP：解析、归一化、校验与状态模型（纯函数，无 Qt 依赖）。"""
from __future__ import annotations

import ipaddress
import re
from dataclasses import dataclass, field

MAX_DIRECT_IPS = 100

_SEPARATORS = re.compile(r"[\s,;、，；]+")
_FULLWIDTH_DIGITS = str.maketrans("０１２３４５６７８９", "0123456789")
_SURROUNDING = " \t\r\n\"'“”‘’()（）[]【】"


def normalize_ip_token(token: str) -> str:
    token = str(token).translate(_FULLWIDTH_DIGITS)
    token = token.replace("．", ".").replace("\u3000", " ")
    token = token.strip().strip(_SURROUNDING).strip()
    if token.endswith("。"):
        token = token[:-1].strip()
    return token


def split_pasted_ips(text: str) -> list[str]:
    return [
        token
        for token in (normalize_ip_token(part) for part in _SEPARATORS.split(str(text)))
        if token
    ]


def parse_ipv4(token: str) -> str | None:
    token = normalize_ip_token(token)
    if not token:
        return None
    try:
        return str(ipaddress.IPv4Address(token))
    except ipaddress.AddressValueError:
        return None


def parse_direct_ips(value: str | list[str] | tuple[str, ...]) -> list[str]:
    """配置/粘贴输入 → 去重后的合法 IPv4 列表；非法项被丢弃。"""
    if isinstance(value, str):
        raw = split_pasted_ips(value)
    else:
        raw = []
        for item in value:
            raw.extend(split_pasted_ips(str(item)))
    result: list[str] = []
    seen: set[str] = set()
    for token in raw:
        ip = parse_ipv4(token)
        if ip and ip not in seen:
            seen.add(ip)
            result.append(ip)
    return result


@dataclass(frozen=True)
class DirectIpIssue:
    index: int
    value: str
    reason: str
    message: str


@dataclass
class DirectIpValidation:
    valid: list[str] = field(default_factory=list)
    skipped_local: list[str] = field(default_factory=list)
    issues: list[DirectIpIssue] = field(default_factory=list)


def _is_reserved(addr: ipaddress.IPv4Address) -> bool:
    return (
        addr.is_unspecified
        or addr.is_loopback
        or addr.is_multicast
        or addr == ipaddress.IPv4Address("255.255.255.255")
    )


def validate_direct_ips(
    values: list[str],
    *,
    forbidden_ips: set[str] | frozenset[str] = frozenset(),
    local_networks: tuple[ipaddress.IPv4Network, ...] = (),
) -> DirectIpValidation:
    result = DirectIpValidation()
    seen: set[str] = set()
    for index, raw in enumerate(values):
        token = normalize_ip_token(raw)
        if not token:
            continue
        ip = parse_ipv4(token)
        if ip is None:
            result.issues.append(
                DirectIpIssue(index, raw, "invalid", "不是合法的 IPv4 地址")
            )
            continue
        if ip in seen:
            result.issues.append(
                DirectIpIssue(index, ip, "duplicate", "重复的 IP")
            )
            continue
        seen.add(ip)
        addr = ipaddress.IPv4Address(ip)
        if _is_reserved(addr):
            result.issues.append(
                DirectIpIssue(index, ip, "reserved", "不能使用保留/回环/组播地址")
            )
            continue
        if ip in forbidden_ips:
            result.issues.append(
                DirectIpIssue(index, ip, "forbidden", "不能排除本机、网关或 VPN 服务器")
            )
            continue
        if any(addr in network for network in local_networks):
            result.skipped_local.append(ip)
            continue
        result.valid.append(ip)
    return result


def ip_set_changed(old: list[str], new: list[str]) -> bool:
    return set(old) != set(new)


def status_for_ip(ip: str, results: dict[str, bool], pending: set[str]) -> str:
    if ip in pending:
        return "pending"
    if ip in results:
        return "ok" if results[ip] else "failed"
    return "unapplied"
```

# app/privileged_helper/routes.py
"""macOS 直连 /32 路由管理（helper 内部使用，可注入 runner 便于单测）。"""
from __future__ import annotations

import ipaddress
import json
import os
import subprocess
from dataclasses import dataclass

STATE_PATH = "/var/run/bitzh-connect-direct-routes.json"


@dataclass
class RouteAttempt:
    ip: str
    ok: bool
    error: str = ""


def parse_default_route(text: str) -> tuple[str | None, str | None]:
    gateway = interface = None
    for line in text.splitlines():
        line = line.strip()
        if line.startswith("gateway:"):
            gateway = line.split(":", 1)[1].strip() or None
        elif line.startswith("interface:"):
            interface = line.split(":", 1)[1].strip() or None
    return gateway, interface


def run_route(args: list[str]) -> subprocess.CompletedProcess:
    return subprocess.run(args, capture_output=True, text=True)


class DirectRouteManager:
    def __init__(self, state_path: str = STATE_PATH, runner=run_route):
        self._state_path = state_path
        self._runner = runner

    def default_gateway(self) -> tuple[str | None, str | None]:
        result = self._runner(["route", "-n", "get", "default"])
        if result.returncode != 0:
            return None, None
        gateway, interface = parse_default_route(result.stdout)
        if not gateway or not interface:
            return gateway, interface
        if interface.startswith(("utun", "tun")):
            return None, interface
        return gateway, interface

    def apply(
        self, ips: list[str]
    ) -> tuple[list[RouteAttempt], str | None, str | None, str | None]:
        if not ips:
            return [], None, None, None
        gateway, interface = self.default_gateway()
        if gateway is None:
            return [], gateway, interface, "no_physical_default_gateway"
        attempts: list[RouteAttempt] = []
        applied: list[str] = []
        for ip in ips:
            result = self._runner(["route", "-n", "add", "-host", ip, gateway])
            if result.returncode == 0:
                attempts.append(RouteAttempt(ip, True))
                applied.append(ip)
                continue
            stderr = (result.stderr or "").strip()
            if "File exists" in stderr:
                check = self._runner(["route", "-n", "get", ip])
                if check.returncode == 0 and f"gateway: {gateway}" in check.stdout:
                    attempts.append(RouteAttempt(ip, True))
                    applied.append(ip)
                else:
                    attempts.append(
                        RouteAttempt(ip, False, "route_exists_other_gateway")
                    )
                continue
            attempts.append(RouteAttempt(ip, False, stderr or "route_add_failed"))
        self._write_state(applied, gateway)
        return attempts, gateway, interface, None

    def cleanup(self, ips: list[str]) -> None:
        for ip in ips:
            try:
                ipaddress.IPv4Address(ip)
            except ipaddress.AddressValueError:
                continue
            self._runner(["route", "-n", "delete", "-host", ip])
        self._clear_state()

    def cleanup_stale(self) -> None:
        data = self._read_state()
        if not data:
            return
        for ip in data.get("ips", []):
            try:
                ipaddress.IPv4Address(ip)
            except ipaddress.AddressValueError:
                continue
            self._runner(["route", "-n", "delete", "-host", ip])
        self._clear_state()

    def _read_state(self) -> dict | None:
        try:
            with open(self._state_path, encoding="utf-8") as f:
                data = json.load(f)
            return data if isinstance(data, dict) else None
        except (OSError, ValueError):
            return None

    def _write_state(self, ips: list[str], gateway: str) -> None:
        try:
            os.makedirs(os.path.dirname(self._state_path), exist_ok=True)
            with open(self._state_path, "w", encoding="utf-8") as f:
                json.dump({"gateway": gateway, "ips": ips}, f)
        except OSError:
            pass

    def _clear_state(self) -> None:
        try:
            os.unlink(self._state_path)
        except OSError:
            pass
```

import socket
import psutil


def local_ipv4_addresses() -> set[str]:
    result = set()
    for addrs in psutil.net_if_addrs().values():
        for addr in addrs:
            if addr.family == socket.AF_INET and addr.address:
                result.add(addr.address)
    return result


def local_ipv4_networks() -> tuple[ipaddress.IPv4Network, ...]:
    result = []
    for addrs in psutil.net_if_addrs().values():
        for addr in addrs:
            if addr.family == socket.AF_INET and addr.address and addr.netmask:
                try:
                    result.append(
                        ipaddress.IPv4Network(
                            f"{addr.address}/{addr.netmask}", strict=False
                        )
                    )
                except ValueError:
                    pass
    return tuple(result)
```

def _parse_route_get_gateway(text: str) -> str | None:
    for line in text.splitlines():
        stripped = line.strip()
        if stripped.startswith("gateway:"):
            return stripped.split(":", 1)[1].strip() or None
    return None


def physical_gateway() -> str | None:
    try:
        if system() == "Darwin":
            out = subprocess.check_output(
                ["route", "-n", "get", "default"],
                text=True,
                stderr=subprocess.DEVNULL,
            )
            return _parse_route_get_gateway(out)
        if system() == "Linux":
            out = subprocess.check_output(
                ["ip", "route", "show", "default"],
                text=True,
                stderr=subprocess.DEVNULL,
            )
            for line in out.splitlines():
                parts = line.split()
                if "via" in parts:
                    return parts[parts.index("via") + 1]
    except Exception:
        return None
    return None
```
